from flask import Flask, request, jsonify import sqlite3 import random app = Flask(__name__) DATABASE = "hospital_op_hub.db" def get_db(): conn = sqlite3.connect(DATABASE) conn.row_factory = sqlite3.Row return conn # Create database table def init_db(): conn = get_db() conn.execute(""" CREATE TABLE IF NOT EXISTS bookings ( id INTEGER PRIMARY KEY AUTOINCREMENT, op_token TEXT UNIQUE, hospital TEXT NOT NULL, doctor TEXT NOT NULL, booking_date TEXT NOT NULL, booking_time TEXT NOT NULL, patient_name TEXT NOT NULL, age INTEGER NOT NULL, gender TEXT NOT NULL, mobile TEXT NOT NULL, reason TEXT, status TEXT DEFAULT 'Pending' ) """) conn.commit() conn.close() # Test backend @app.route("/") def home(): return "Hospital OP Hub Backend is Running!" # Save OP booking @app.route("/api/book", methods=["POST"]) def book_op(): data = request.get_json() hospital = data.get("hospital") doctor = data.get("doctor") booking_date = data.get("date") booking_time = data.get("time") patient_name = data.get("patientName") age = data.get("age") gender = data.get("gender") mobile = data.get("mobile") reason = data.get("reason", "") if not all([ hospital, doctor, booking_date, booking_time, patient_name, age, gender, mobile ]): return jsonify({ "success": False, "message": "Required details missing" }), 400 op_token = "OP" + str(random.randint(100000, 999999)) conn = get_db() conn.execute(""" INSERT INTO bookings ( op_token, hospital, doctor, booking_date, booking_time, patient_name, age, gender, mobile, reason ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?) """, ( op_token, hospital, doctor, booking_date, booking_time, patient_name, age, gender, mobile, reason )) conn.commit() conn.close() return jsonify({ "success": True, "message": "OP booking saved successfully", "opToken": op_token, "status": "Pending" }) # View all bookings @app.route("/api/bookings", methods=["GET"]) def get_bookings(): conn = get_db() bookings = conn.execute(""" SELECT * FROM bookings ORDER BY id DESC """).fetchall() conn.close() return jsonify([ dict(row) for row in bookings ]) if __name__ == "__main__": init_db() app.run( debug=True, host="127.0.0.1", port=5000 )
